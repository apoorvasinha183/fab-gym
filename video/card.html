pub fn key(c: &Cand, x: &Ctx) -> f64 {
    let lots = c.lots.max(1.0);
    let per_lot = c.proc_minutes / lots;
    let q = x.queue;
    let idle = x.idle;
    let is_bneck = q > 11.0 && idle < 1.0;

    let mut k = 0.0;
    // throughput costs
    k += c.setup_change * 70.0;
    k += per_lot * 0.60;
    k += c.work_left_minutes.ln_1p() * 0.75;
    k += c.steps_left * 0.32;
    k += c.size_wafers * 0.02;
    if c.family == 2.0 {
        k += 9.5;
    }
    // flow benefits
    k -= c.lots * 28.0;
    k -= c.is_batch * 16.0;
    k -= c.priority * 72.0;
    k -= c.waited_minutes * 0.14;
    k -= c.age_minutes * 0.005;

    // very light CQT, only imminent breach
    let s = c.cqt_slack_minutes;
    if s < 1e8 && s < 35.0 {
        let mut b = (35.0 - s) * 0.60;
        if b > 50.0 { b = 50.0; }
        k -= b;
    }

    // polishing bottleneck adaptation
    if is_bneck {
        // at bottleneck maximize throughput: avoid setup, prefer full batches, SPT
        k += c.setup_change * 16.0;
        k -= c.lots * 7.0;
        k += per_lot * 0.15;
        k += c.work_left_minutes.ln_1p() * 0.20;
        // scale slightly with queue depth
        k += c.setup_change * q * 0.35;
        k -= c.lots * q * 0.08;
    } else {
        // upstream/downstream keep flow to bottleneck
        k -= c.waited_minutes * 0.05;
    }

    // anti-starvation for very long waiters
    if c.waited_minutes > 340.0 {
        k -= (c.waited_minutes - 340.0) * 0.028;
    }

    if c.is_default == 1.0 {
        k -= 0.0008;
    }
    if !k.is_finite() {
        return 0.0;
    }
    k
}
